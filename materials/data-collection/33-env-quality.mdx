<PyPlayground
  client:visible
  title="环境数据质控平台：全流程一次跑通"
  height={360}
  note="试着把西站故障值 9.99 改成 0.05（正常值），看质量评分与审计日志如何变化；再试着把审计日志删掉一行——体会「没有审计的清洗」为什么不可信。"
  code={`import statistics as st

east = [("08:00", 35.2), ("09:00", 38.1), ("10:00", None),
        ("11:00", 41.5), ("12:00", None), ("13:00", 39.8)]   # μg/m3
west = [("08:00", 0.033), ("09:00", 0.036), ("10:00", 9.99),
        ("11:00", 0.040), ("12:00", 0.038), ("13:00", 0.037)]  # mg/m3！

audit = []

# ① 单位转换：西站 mg/m3 → μg/m3
west_ug = [(t, v * 1000) for t, v in west]
audit.append("转换：西站单位 mg/m3 -> μg/m3（×1000）")

# ② 异常隔离：3σ 边界外的值标记隔离，不删除
vals = [v for _, v in west_ug]
mean, std = st.mean(vals), st.pstdev(vals)
quarantine = [(t, v) for t, v in west_ug if abs(v - mean) > 3 * std]
west_ok = [(t, v) for t, v in west_ug if abs(v - mean) <= 3 * std]
audit.append(f"清洗：西站隔离异常 {len(quarantine)} 条（进隔离区，原始保留）")

# ③ 缺失填充：东站 None 用中位数补，并标记
med = st.median(v for _, v in east if v is not None)
east_ok = [(t, v if v is not None else med) for t, v in east]
n_fill = sum(1 for _, v in east if v is None)
audit.append(f"填充：东站缺失 {n_fill} 条用中位数 {med} 补齐（已标记）")

# ④ 质量评估：完整性 / 准确性 / 一致性
integ = 1 - n_fill / len(east)
accu = 1 - len(quarantine) / len(west)
score = round((integ + accu + 1.0) / 3 * 100, 1)   # 一致性：单位已统一记 1.0
audit.append(f"评估：完整性 {integ:.2f} 准确性 {accu:.2f} 一致性 1.00")

print(f"东站(处理后): {[(t, round(v,1)) for t, v in east_ok]}")
print(f"西站(处理后): {[(t, round(v,1)) for t, v in west_ok]}")
print(f"隔离区: {quarantine}")
print(f"综合质量评分: {score} / 100")
print()
print("=== 操作审计（上报材料附件） ===")
for i, line in enumerate(audit, 1):
    print(f"{i}. {line}")`}
